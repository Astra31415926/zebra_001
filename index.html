// ════════════════════════════════════════════════════════ // ZEBRA FINDER v6.3 — findFrame() з multi-candidate ranking // ════════════════════════════════════════════════════════ // Shrink квадрилатераля на px пікселів (з урахуванням перспективи) function shrinkQuad(c, px) { // Центр квадрилатераля const cx = (c.tl.x + c.tr.x + c.br.x + c.bl.x) / 4; const cy = (c.tl.y + c.tr.y + c.br.y + c.bl.y) / 4; function moveToward(p, factor) { return { x: Math.round(p.x + (cx - p.x) * factor), y: Math.round(p.y + (cy - p.y) * factor) }; } // Оцінюємо розмір квадрилатераля для розрахунку відносного shrink const diag = Math.hypot(c.br.x - c.tl.x, c.br.y - c.tl.y); const factor = diag > 0 ? px / (diag / 2) : 0; return { tl: moveToward(c.tl, factor), tr: moveToward(c.tr, factor), br: moveToward(c.br, factor), bl: moveToward(c.bl, factor) }; } // Перевірка геометрії кандидата function validateGeometry(c, w, h) { const minDim = Math.min(w, h) * 0.08; const sides = [ Math.hypot(c.tr.x - c.tl.x, c.tr.y - c.tl.y), Math.hypot(c.br.x - c.tr.x, c.br.y - c.tr.y), Math.hypot(c.bl.x - c.br.x, c.bl.y - c.br.y), Math.hypot(c.tl.x - c.bl.x, c.tl.y - c.bl.y) ]; if (sides.some(s => s < minDim)) return false; // Виродженість: площа function cross2d(ax, ay, bx, by) { return ax * by - ay * bx; } const area = 0.5 * Math.abs( cross2d(c.tr.x - c.tl.x, c.tr.y - c.tl.y, c.bl.x - c.tl.x, c.bl.y - c.tl.y) + cross2d(c.bl.x - c.tl.x, c.bl.y - c.tl.y, c.br.x - c.tl.x, c.br.y - c.tl.y) ); if (area < minDim * minDim) return false; // Співвідношення сторін const maxS = Math.max(...sides), minS = Math.min(...sides); if (maxS / minS > 8) return false; return true; } // Швидкий warp кандидата в малий розмір і підрахунок zebra score function quickZebraScore(idata, sw, sh, corners, QS) { // warp в QS×QS const warped = warpPerspective(idata, sw, sh, corners, QS); if (!warped) return 0; const wGray = toGray(warped, QS, QS); const res = verifyZebra(wGray, QS, 40); if (!res) return 0; return res.confidence; } // Нова findFrame з multi-candidate ranking function findFrame(idata, w, h) { const gray = toGray(idata, w, h); const bg = sampleBG(gray, w, h, 5); Li('BG luma: ' + bg); const QS = 180; // розмір для quick warp перевірки const candidates = []; // ── ЕТАП A: генерація кандидатів ── const tols = [15, 20, 30, 42, 55, 70, 90]; for (const tol of tols) { const rm = makeMask(gray, w, h, bg, tol); const em = erodeMask(rm, w, h); const fgN = em.reduce((s, v) => s + v, 0); if (fgN < 50) continue; const c = findCorners(em, w, h, 15); const sp = spread(c); if (sp < 20) continue; if (!validateGeometry(c, w, h)) continue; candidates.push({ corners: c, sp, label: 'tol=' + tol }); // Shrink-кандидати від кожного базового (для пошуку внутрішньої рамки) const shrinkPx = [4, 8, 12, 16, 20, 30, 40, 50]; for (const px of shrinkPx) { const sc = shrinkQuad(c, px); if (!validateGeometry(sc, w, h)) continue; const ssp = spread(sc); if (ssp < 20) continue; candidates.push({ corners: sc, sp: ssp, label: 'tol=' + tol + '+shrink' + px }); } } // Variance-scan fallback { const topRows = [], leftCols = []; for (let y = 0; y < h; y++) { let sum = 0, sum2 = 0; for (let x = 0; x < w; x++) { const v = gray[y * w + x]; sum += v; sum2 += v * v; } const avg = sum / w, std = Math.sqrt(sum2 / w - avg * avg); if (std > 50) topRows.push(y); } for (let x = 0; x < w; x++) { let sum = 0, sum2 = 0; for (let y = 0; y < h; y++) { const v = gray[y * w + x]; sum += v; sum2 += v * v; } const avg = sum / h, std = Math.sqrt(sum2 / h - avg * avg); if (std > 50) leftCols.push(x); } topRows.sort((a, b) => a - b); leftCols.sort((a, b) => a - b); if (topRows.length > 0 && leftCols.length > 0) { const vc = { tl: { x: leftCols[0], y: topRows[0] }, tr: { x: leftCols[leftCols.length - 1], y: topRows[0] }, br: { x: leftCols[leftCols.length - 1], y: topRows[topRows.length - 1] }, bl: { x: leftCols[0], y: topRows[topRows.length - 1] } }; const vsp = spread(vc); if (vsp >= 20 && validateGeometry(vc, w, h)) { candidates.push({ corners: vc, sp: vsp, label: 'variance' }); for (const px of [8, 16, 24, 36]) { const sc = shrinkQuad(vc, px); if (validateGeometry(sc, w, h)) { candidates.push({ corners: sc, sp: spread(sc), label: 'variance+shrink' + px }); } } } } } Li('Кандидатів згенеровано: ' + candidates.length); if (candidates.length === 0) { Le('Жодного кандидата не знайдено'); return { corners: null, spread: 0 }; } // ── ЕТАП B: дедуплікація (за позицією TL) ── const unique = []; for (const cand of candidates) { const dup = unique.some(u => Math.hypot(u.corners.tl.x - cand.corners.tl.x, u.corners.tl.y - cand.corners.tl.y) < 8 && Math.hypot(u.corners.br.x - cand.corners.br.x, u.corners.br.y - cand.corners.br.y) < 8 ); if (!dup) unique.push(cand); } Li('Після дедуплікації: ' + unique.length); // ── ЕТАП C: zebra scoring кожного кандидата ── const imageArea = w * h; const scored = []; Lh('CANDIDATES'); for (const cand of unique) { const c = cand.corners; const area = 0.5 * Math.abs( (c.tr.x - c.tl.x) * (c.bl.y - c.tl.y) - (c.bl.x - c.tl.x) * (c.tr.y - c.tl.y) + (c.br.x - c.tr.x) * (c.tl.y - c.tr.y) - (c.tl.x - c.tr.x) * (c.br.y - c.tr.y) ); const areaPct = area / imageArea; const zScore = quickZebraScore(idata, w, h, c, QS); // Geometry score: штрафуємо занадто великі кандидати без zebra const areaBonus = areaPct > 0.9 ? -0.2 : 0; // штраф якщо займає майже весь ROI // Фінальний score: zebra має максимальний вес const finalScore = zScore * 0.85 + (cand.sp / Math.max(w, h)) * 0.10 + areaBonus; const status = zScore > 0.3 ? 'ACCEPT' : 'REJECT'; Li(` [${cand.label}] area=${(areaPct * 100).toFixed(0)}% zebra=${zScore.toFixed(2)} sp=${Math.round(cand.sp)} score=${finalScore.toFixed(3)} ${status}`); scored.push({ ...cand, zScore, finalScore, areaPct }); } // ── ЕТАП D: вибір найкращого ── scored.sort((a, b) => b.finalScore - a.finalScore); const best = scored[0]; if (!best || best.zScore < 0.05) { // Zebra score занадто низький у всіх — fallback до найбільшого spread Lw('Zebra score низький у всіх — fallback до spread'); scored.sort((a, b) => b.sp - a.sp); const fallback = scored[0]; Li('Fallback: ' + fallback.label + ' spread=' + Math.round(fallback.sp)); return { corners: fallback.corners, spread: fallback.sp }; } Lok('Обрано: ' + best.label + ' zebra=' + best.zScore.toFixed(2) + ' score=' + best.finalScore.toFixed(3) + ' spread=' + Math.round(best.sp)); return { corners: best.corners, spread: best.sp }; }
