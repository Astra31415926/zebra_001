// ═══════════════════════════════════════════════ // verifyZebra() — УСИЛЕНА // Теперь проверяет ПЕРИМЕТРАЛЬНЫЕ линии (offsetы 0..2 от края) // и требует согласия минимум по 3 из 4 сторон. // Сигнатура не изменилась — run() не трогаем. // ═══════════════════════════════════════════════ function verifyZebra(wGray, S, minContrast) { // scanLine: анализирует одну линию пикселей, возвращает {T, modSize} или null function scanLine(arr) { const n = arr.length; let mn = 255, mx = 0; for (const v of arr) { if (v < mn) mn = v; if (v > mx) mx = v; } if (mx - mn < minContrast) return null; const thr = (mn + mx) >> 1, runs = []; let cur = arr[0] > thr ? 1 : 0, len = 1; for (let i = 1; i < n; i++) { const b = arr[i] > thr ? 1 : 0; if (b === cur) len++; else { runs.push({ v: cur, len }); cur = b; len = 1; } } runs.push({ v: cur, len }); if (runs.length < 3) return null; const allLens = runs.map(r => r.len).sort((a, b) => a - b); const med = allLens[allLens.length >> 1]; if (med < 2) return null; const valid = runs.filter(r => r.len >= med * 0.4 && r.len <= med * 2.4); if (valid.length < 5) return null; let T = valid.length; if (T % 2 === 0) { if (valid[0].v === 1 || valid[T - 1].v === 1) T += 1; else return null; } if (T < 5) return null; return { T, modSize: S / T }; } const votes = new Map(); let total = 0; // --- СТАРЫЕ пробы (внутренние офсеты) --- const innerOffsets = [2, 5, 10, 15, 20, 30]; for (const off of innerOffsets) { if (off >= S / 2) continue; for (const pos of [off, S - 1 - off]) { const row = new Uint8Array(S); for (let x = 0; x < S; x++) row[x] = wGray[pos * S + x]; const rh = scanLine(row); if (rh) votes.set(rh.T, (votes.get(rh.T) || 0) + 1); total++; const col = new Uint8Array(S); for (let y = 0; y < S; y++) col[y] = wGray[y * S + pos]; const rv = scanLine(col); if (rv) votes.set(rv.T, (votes.get(rv.T) || 0) + 1); total++; } } if (!votes.size) return null; // --- НОВАЯ ЧАСТЬ: проверка по 4 сторонам периметра --- // Для настоящей TAINA рамки: // - Строки y=0..2 и y=S-3..S-1 должны давать T-голоса (горизонтальная зебра) // - Столбцы x=0..2 и x=S-3..S-1 должны давать T-голоса (вертикальная зебра) // Требуем: хотя бы 3 из 4 сторон согласны с bestT const sorted = [...votes.entries()].sort((a, b) => b[1] - a[1]); const [bestT, bestV] = sorted[0]; // Сканируем по 3 крайних строки/столбца с каждой стороны const EDGE_LINES = 3; let sidesOk = 0; // Топ (строки 0..EDGE_LINES-1) { let ok = 0; for (let y = 0; y < EDGE_LINES; y++) { const row = new Uint8Array(S); for (let x = 0; x < S; x++) row[x] = wGray[y * S + x]; const r = scanLine(row); if (r && r.T === bestT) ok++; } if (ok >= 2) sidesOk++; } // Низ (строки S-EDGE_LINES..S-1) { let ok = 0; for (let y = S - EDGE_LINES; y < S; y++) { const row = new Uint8Array(S); for (let x = 0; x < S; x++) row[x] = wGray[y * S + x]; const r = scanLine(row); if (r && r.T === bestT) ok++; } if (ok >= 2) sidesOk++; } // Левый (столбцы 0..EDGE_LINES-1) { let ok = 0; for (let x = 0; x < EDGE_LINES; x++) { const col = new Uint8Array(S); for (let y = 0; y < S; y++) col[y] = wGray[y * S + x]; const r = scanLine(col); if (r && r.T === bestT) ok++; } if (ok >= 2) sidesOk++; } // Правый (столбцы S-EDGE_LINES..S-1) { let ok = 0; for (let x = S - EDGE_LINES; x < S; x++) { const col = new Uint8Array(S); for (let y = 0; y < S; y++) col[y] = wGray[y * S + x]; const r = scanLine(col); if (r && r.T === bestT) ok++; } if (ok >= 2) sidesOk++; } // Требуем минимум 3 стороны из 4 (для надёжности, а не ложной рамки) if (sidesOk < 3) return null; const vStr = sorted.slice(0, 5).map(([t, v]) => `T${t}:${v}`).join(' '); return { T: bestT, modSize: S / bestT, confidence: bestV / total, voteStr: vStr, sidesOk }; } // ═══════════════════════════════════════════════ // findFrameWithZebra() — НОВАЯ ФУНКЦИЯ // Объединяет: поиск кандидатов + WARP + Zebra в одном проходе. // Возвращает {corners, warped, wGray, S, zebra} при успехе, // или {corners: null} если ни один кандидат не прошёл Zebra. // // Используется вместо findFrame() + warpPerspective() + verifyZebra() // в run() и scanTick(). // ═══════════════════════════════════════════════ function findFrameWithZebra(gray, w, h, idata, S = 600, silent = false) { // Шаг 1: собрать кандидатов по нескольким гипотезам масок const bx = Math.max(1, Math.round(w * 0.08)); const by_ = Math.max(1, Math.round(h * 0.08)); const perim = []; for (let y = 0; y < by_; y++) for (let x = 0; x < w; x += 2) perim.push(gray[y * w + x]); for (let y = h - by_; y < h; y++) for (let x = 0; x < w; x += 2) perim.push(gray[y * w + x]); for (let y = by_; y < h - by_; y += 2) for (let x = 0; x < bx; x++) perim.push(gray[y * w + x]); for (let y = by_; y < h - by_; y += 2) for (let x = w - bx; x < w; x++) perim.push(gray[y * w + x]); perim.sort((a, b) => a - b); const bgPerim = perim[perim.length >> 1]; const hist = new Int32Array(256); for (let i = 0; i < w * h; i++) hist[gray[i]]++; const N = w * h; let sum = 0; for (let i = 0; i < 256; i++) sum += i * hist[i]; let wb_ = 0, sb_ = 0, bestVOtsu = 0, otsuT = 128; for (let t = 0; t < 256; t++) { wb_ += hist[t]; if (!wb_) continue; const wf = N - wb_; if (!wf) break; sb_ += t * hist[t]; const mb = sb_ / wb_, mf = (sum - sb_) / wf, v = wb_ * wf * (mb - mf) * (mb - mf); if (v > bestVOtsu) { bestVOtsu = v; otsuT = t; } } // Строим несколько масок (как в findFrameGray) const tols = [15, 25, 35, 50, 70]; const candidates = []; // {corners, score, label} const tryMask = (mask, label) => { const em = erodeMask(mask, w, h); const fgN = em.reduce((s, v) => s + v, 0); if (fgN < 100) return; const c = findCorners(em, w, h, 15); const sc = scoreCandidate(c, w, h, true); // silent=true — не спамить в лог if (sc > 0) candidates.push({ corners: c, score: sc, label }); }; for (const tol of tols) { const mask = makeMask(gray, w, h, bgPerim, tol); tryMask(mask, 'H1-bg' + bgPerim + '-tol' + tol); } const bgInv = 255 - bgPerim; for (const tol of tols) { const mask = makeMask(gray, w, h, bgInv, tol); tryMask(mask, 'H2-bgInv-tol' + tol); } { const mask = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) mask[i] = gray[i] < otsuT ? 1 : 0; tryMask(mask, 'H3-otsuDark'); } { const mask = new Uint8Array(w * h); for (let i = 0; i < w * h; i++) mask[i] = gray[i] > otsuT ? 1 : 0; tryMask(mask, 'H4-otsuLight'); } { const mask = new Uint8Array(w * h); for (let y = 1; y < h - 1; y++) for (let x = 1; x < w - 1; x++) { let mn = 255, mx = 0; for (let dy = -1; dy <= 1; dy++) for (let dx = -1; dx <= 1; dx++) { const v = gray[(y + dy) * w + (x + dx)]; if (v < mn) mn = v; if (v > mx) mx = v; } mask[y * w + x] = mx - mn > 40 ? 1 : 0; } tryMask(mask, 'H5-edge'); } if (!candidates.length) { if (!silent) Le(' findFrameWithZebra: нет геометрических кандидатов'); return { corners: null }; } // Дедупликация: убираем похожих кандидатов (одинаковые corners с допуском) const unique = []; outer: for (const cand of candidates) { for (const u of unique) { if (cornersClose(cand.corners, u.corners, Math.min(w, h))) continue outer; } unique.push(cand); } // Сортируем по score (лучший первым) unique.sort((a, b) => b.score - a.score); if (!silent) Li(' кандидатів: ' + candidates.length + ' → uniq: ' + unique.length); // Шаг 2: для каждого уникального кандидата — WARP → Zebra const exp = 6; for (const cand of unique) { const c = cand.corners; const expanded = { tl: { x: c.tl.x - exp, y: c.tl.y - exp }, tr: { x: c.tr.x + exp, y: c.tr.y - exp }, br: { x: c.br.x + exp, y: c.br.y + exp }, bl: { x: c.bl.x - exp, y: c.bl.y + exp } }; const warped = warpPerspective(idata, w, h, expanded, S); if (!warped) { if (!silent) Li(' [' + cand.label + '] warp failed — skip'); continue; } const wGray = toGray(warped, S, S); const zebra = verifyZebra(wGray, S, 50); if (!silent) { if (zebra) { Lok(' [' + cand.label + '] score=' + cand.score.toFixed(0) + ' → ZEBRA OK: T=' + zebra.T + ' sides=' + zebra.sidesOk + ' ' + zebra.voteStr); } else { Li(' [' + cand.label + '] score=' + cand.score.toFixed(0) + ' → Zebra FAIL'); } } if (zebra) { // Нашли! Возвращаем всё нужное для run() return { corners: expanded, warped, wGray, S, zebra }; } } if (!silent) Le(' findFrameWithZebra: ні один кандидат не пройшов Zebra'); return { corners: null }; } // ═══════════════════════════════════════════════ // run() — ИЗМЕНЕНО // Теперь вместо findFrame() + warpPerspective() + verifyZebra() по отдельности — // вызывается findFrameWithZebra(), которая делает всё в правильном порядке. // Остальное (шаги 4-5) не изменено. // ═══════════════════════════════════════════════ function run(idata, w, h) { // ── Шаги 1+2+3 объединены в findFrameWithZebra() ── Lh('1–3. КАНДИДАТ → WARP → ZEBRA'); const gray = toGray(idata, w, h); const result = findFrameWithZebra(gray, w, h, idata, 600, false); if (!result.corners) { Le('Рамку з Zebra не знайдено'); drawResult(null, null, null, idata, w, h, null); setSt('✗ рамку не знайдено', false); showResult(null, null); return; } const { corners, warped, wGray, S, zebra } = result; Li('TL(' + corners.tl.x + ',' + corners.tl.y + ') TR(' + corners.tr.x + ',' + corners.tr.y + ') BR(' + corners.br.x + ',' + corners.br.y + ') BL(' + corners.bl.x + ',' + corners.bl.y + ')'); Lok('Warp → ' + S + '×' + S); if (zebra) Li(' голоси: ' + zebra.voteStr); Lok('T=' + zebra.T + ' mod=' + zebra.modSize.toFixed(1) + 'px conf=' + (zebra.confidence * 100).toFixed(0) + '% sides=' + zebra.sidesOk); Lh('4. КРУЖЕЧКИ'); const circles = sampleCircles(warped, S, zebra.T, zebra.modSize); const lumas = circles.cells.map(c => (c.r + c.g + c.b) / 3).sort((a, b) => a - b); Li(' ' + circles.cells.length + ' шт r=' + circles.circleR.toFixed(1) + 'px'); Li(' яскравість: min=' + lumas[0].toFixed(0) + ' med=' + lumas[lumas.length >> 1].toFixed(0) + ' max=' + lumas[lumas.length - 1].toFixed(0)); Lh('5. ДЕКОДУВАННЯ'); const decoded = decodeCircles(circles, zebra.T); drawResult(warped, S, zebra, idata, w, h, corners); buildRGBPanel(warped, S, zebra.T, zebra.modSize, circles); if (decoded && decoded.text !== null) { Lok('РЕЗУЛЬТАТ: "' + decoded.text + '" [' + decoded.type + ']'); setSt('✓ "' + decoded.text + '"', true); showResult(decoded.text, decoded.type); } else { Lw('Декод не вдався [' + (decoded ? decoded.type : '—') + ']'); setSt('⚠ зебра OK, декод null', null); showResult(null, decoded ? decoded.type : null); } } // ═══════════════════════════════════════════════ // scanTick() — ИЗМЕНЕНО // Вместо findFrameGray() + temporal + отдельный run() — // теперь findFrameWithZebra() в silent-режиме для скорости. // Temporal stability остаётся: нужно 2 последовательных кадра // с успешной Zebra И близкими corners. // ═══════════════════════════════════════════════ function scanTick() { if (!stream || !vid.videoWidth) return; const elapsed = Date.now() - scanStartTime; if (elapsed >= SCAN_TIMEOUT) { scanLoopStop(); timerBarWrap.style.display = 'none'; setSt('⚠ час вийшов — наведи знову', null); return; } const vw = vid.videoWidth, vh = vid.videoHeight; const roiSize = Math.min(vw, vh) * 0.7; const sx = (vw - roiSize) / 2, sy = (vh - roiSize) / 2; if (!roiCanvas) { roiCanvas = document.createElement('canvas'); } const side = Math.min(600, Math.round(roiSize)); if (roiCanvas.width !== side || roiCanvas.height !== side) { roiCanvas.width = side; roiCanvas.height = side; } roiCanvas.getContext('2d').drawImage(vid, sx, sy, roiSize, roiSize, 0, 0, side, side); // raw grayscale для detection const { gray, w, h } = getGrayFromCanvas(roiCanvas, 600); // RAW idata (не Otsu) нужна для warpPerspective внутри findFrameWithZebra // Используем тот же roiCanvas — getImageData напрямую const rawIdata = roiCanvas.getContext('2d').getImageData(0, 0, w, h); // findFrameWithZebra в silent-режиме (не спамит лог каждые 250ms) const result = findFrameWithZebra(gray, w, h, rawIdata, 400, true); // S=400 для скорости в реальном времени (меньше pixels в warp) if (!result.corners) { stableCount = 0; lastGoodCorners = null; return; } // Temporal stability: corners должны совпадать 2 кадра подряд const minSide = Math.min(w, h); if (lastGoodCorners && cornersClose(result.corners, lastGoodCorners, minSide)) { stableCount++; } else { stableCount = 1; } lastGoodCorners = result.corners; if (stableCount < STABLE_NEEDED) return; // Стабильная рамка с подтверждённой Zebra — запускаем декод scanLoopStop(); timerBarWrap.style.display = 'none'; vidWrap.style.display = 'none'; mainCnv.style.display = 'block'; clrLog(); setSt('обробка…', null); // Для финального декода берём более качественный кадр (900px) const decCnv = document.createElement('canvas'); const decSize = Math.min(900, Math.round(roiSize)); decCnv.width = decSize; decCnv.height = decSize; decCnv.getContext('2d').drawImage(vid, sx, sy, roiSize, roiSize, 0, 0, decSize, decSize); const { idata, w: dw, h: dh } = prepareCanvasForDecode(decCnv); run(idata, dw, dh); }
